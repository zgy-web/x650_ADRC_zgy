cmake_minimum_required(VERSION 3.16)
project(flyarm_mavlink_uart LANGUAGES CXX)
set(MAVLINK_INCLUDE_DIR "" CACHE PATH "Pinned generated development dialect")
if(NOT EXISTS "${MAVLINK_INCLUDE_DIR}/development/mavlink.h")
  message(FATAL_ERROR "Generate the audited dialect before building")
endif()
add_executable(flyarm_uart_bench src/bench.cpp)
add_executable(test_mavlink_contract tests/test_contract.cpp)
add_executable(flyarm_uart_bench_pty src/bench.cpp)
target_compile_definitions(flyarm_uart_bench_pty PRIVATE FLYARM_UART_TEST_DEVICE=1)
foreach(target flyarm_uart_bench test_mavlink_contract flyarm_uart_bench_pty)
  target_include_directories(${target} PRIVATE include)
  target_include_directories(${target} SYSTEM PRIVATE "${MAVLINK_INCLUDE_DIR}")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME mavlink_contract COMMAND test_mavlink_contract)
