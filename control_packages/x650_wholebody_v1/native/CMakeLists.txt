cmake_minimum_required(VERSION 3.16)
project(x650_optimization LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(x650_optimization SHARED optimization.cpp vendor/quadprogpp/QuadProg++.cc)
target_link_libraries(x650_optimization PRIVATE Eigen3::Eigen)
target_compile_features(x650_optimization PRIVATE cxx_std_17)
target_compile_options(x650_optimization PRIVATE -Wall -Wextra)
add_library(x650_mpc_core SHARED mpc_core.cpp)
target_include_directories(x650_mpc_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(x650_mpc_core PUBLIC x650_optimization PRIVATE Eigen3::Eigen)
target_compile_features(x650_mpc_core PUBLIC cxx_std_17)
target_compile_options(x650_mpc_core PRIVATE -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_mpc_core test_mpc_core.cpp ../x650_wholebody/ts20_prediction_native.cpp)
  target_link_libraries(test_mpc_core PRIVATE x650_mpc_core)
  target_compile_options(test_mpc_core PRIVATE -Wall -Wextra)
  add_test(NAME mpc_core_raw_input_transaction COMMAND test_mpc_core)
  add_executable(test_interval_core test_interval_core.cpp ../x650_wholebody/ts20_prediction_native.cpp)
  target_link_libraries(test_interval_core PRIVATE x650_mpc_core)
  target_compile_features(test_interval_core PRIVATE cxx_std_17)
  target_compile_options(test_interval_core PRIVATE -Wall -Wextra)
  add_test(NAME mpc_interval_native_transaction COMMAND test_interval_core)
  add_executable(test_interval_observer test_interval_observer.cpp ../x650_wholebody/ts20_prediction_native.cpp)
  target_link_libraries(test_interval_observer PRIVATE x650_mpc_core)
  target_compile_features(test_interval_observer PRIVATE cxx_std_17)
  target_compile_options(test_interval_observer PRIVATE -Wall -Wextra)
  add_test(NAME mpc_interval_canonical_observer_and_hard_rows COMMAND test_interval_observer)
  add_executable(test_quadprog_identity test_quadprog_identity.cpp vendor/quadprogpp/QuadProg++.cc)
  target_compile_features(test_quadprog_identity PRIVATE cxx_std_17)
  target_compile_options(test_quadprog_identity PRIVATE -Wall -Wextra)
  add_test(NAME quadprog_identity_initialization COMMAND test_quadprog_identity)
endif()
