cmake_minimum_required(VERSION 3.16)
project(x650_upper_nmpc_offline VERSION 0.1.0 LANGUAGES CXX)
set(X650_BACKEND_LIBRARY_DIR "" CACHE PATH "Existing ABI4/ABI2 library directory")
set(X650_TS20_LIBRARY "" CACHE FILEPATH "Existing libts20_prediction_v3.so")
set(X650_NATIVE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../x650_wholebody_v1/native" CACHE PATH "Existing matching ABI4 headers")
if(NOT EXISTS "${X650_BACKEND_LIBRARY_DIR}/libx650_mpc_core.so" OR NOT EXISTS "${X650_TS20_LIBRARY}" OR NOT EXISTS "${X650_BACKEND_LIBRARY_DIR}/libx650_optimization.so")
  message(FATAL_ERROR "Existing core, optimization and predictor libraries required")
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(OpenSSL REQUIRED)
add_library(x650_nmpc_offline STATIC src/objective.cpp src/offline_solver.cpp)
target_compile_features(x650_nmpc_offline PUBLIC cxx_std_17)
target_compile_options(x650_nmpc_offline PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(x650_nmpc_offline PUBLIC include "${X650_NATIVE_INCLUDE_DIR}")
target_link_libraries(x650_nmpc_offline PUBLIC Eigen3::Eigen
  "${X650_BACKEND_LIBRARY_DIR}/libx650_mpc_core.so"
  "${X650_BACKEND_LIBRARY_DIR}/libx650_optimization.so" "${X650_TS20_LIBRARY}")
add_executable(x650_nmpc_offline_replay examples/offline_replay.cpp)
target_include_directories(x650_nmpc_offline_replay PRIVATE ../x650_wholebody_v1/benchmark/include)
target_link_libraries(x650_nmpc_offline_replay PRIVATE x650_nmpc_offline PkgConfig::JSONCPP OpenSSL::Crypto ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING)
  add_executable(test_x650_nmpc_offline tests/test_solver.cpp)
  target_include_directories(test_x650_nmpc_offline PRIVATE ../x650_mpc_controller_v1/examples ../x650_mpc_controller_v1/include ../x650_wholebody_v1/benchmark/include)
  target_link_libraries(test_x650_nmpc_offline PRIVATE x650_nmpc_offline PkgConfig::JSONCPP OpenSSL::Crypto)
  add_test(NAME nmpc_offline_transaction COMMAND test_x650_nmpc_offline)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_objective.cpp")
    add_executable(test_x650_nmpc_objective tests/test_objective.cpp)
    target_include_directories(test_x650_nmpc_objective PRIVATE ../x650_mpc_controller_v1/examples ../x650_mpc_controller_v1/include)
    target_link_libraries(test_x650_nmpc_objective PRIVATE x650_nmpc_offline)
    add_test(NAME nmpc_nonlinear_objective COMMAND test_x650_nmpc_objective)
  endif()
endif()
