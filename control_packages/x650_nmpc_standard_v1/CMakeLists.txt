cmake_minimum_required(VERSION 3.16)
project(x650_standard_nmpc LANGUAGES C CXX)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(OpenSSL REQUIRED)
find_package(Eigen3 REQUIRED)
add_subdirectory(../x650_wholebody_v1/native legacy_native)
add_library(x650_standard_case_builder STATIC native/case_builder.cpp)
target_include_directories(x650_standard_case_builder PUBLIC native ../x650_wholebody_v1/native)
target_compile_features(x650_standard_case_builder PUBLIC cxx_std_17)
target_link_libraries(x650_standard_case_builder PUBLIC Eigen3::Eigen)
add_executable(x650_standard_case_replay native/case_builder_replay.cpp)
target_include_directories(x650_standard_case_replay PRIVATE ../x650_wholebody_v1/benchmark/include)
target_link_libraries(x650_standard_case_replay PRIVATE x650_standard_case_builder PkgConfig::JSONCPP OpenSSL::Crypto)
add_library(x650_standard_event_predictor SHARED ../x650_wholebody_v1/x650_wholebody/ts20_prediction_native.cpp)
target_compile_features(x650_standard_event_predictor PRIVATE cxx_std_17)
add_executable(x650_standard_event_replay native/event_replay.cpp)
target_include_directories(x650_standard_event_replay PRIVATE ../x650_wholebody_v1/benchmark/include)
target_compile_features(x650_standard_event_replay PRIVATE cxx_std_17)
target_link_libraries(x650_standard_event_replay PRIVATE x650_mpc_core x650_standard_event_predictor PkgConfig::JSONCPP OpenSSL::Crypto)
if(BUILD_TESTING)
  add_executable(test_standard_event_validator tests/test_event_validator.cpp)
  target_include_directories(test_standard_event_validator PRIVATE ../x650_mpc_controller_v1/examples ../x650_mpc_controller_v1/include)
  target_compile_features(test_standard_event_validator PRIVATE cxx_std_17)
  target_link_libraries(test_standard_event_validator PRIVATE x650_mpc_core x650_standard_event_predictor)
  add_test(NAME common_event_candidate_validator COMMAND test_standard_event_validator)
endif()

set(X650_ACADOS_GENERATED_DIR "" CACHE PATH "Directory containing generated x650_standard solver")
set(ACADOS_SOURCE_DIR "$ENV{ACADOS_SOURCE_DIR}" CACHE PATH "Pinned acados installation")
if(X650_ACADOS_GENERATED_DIR)
  if(NOT EXISTS "${X650_ACADOS_GENERATED_DIR}/acados_solver_x650_standard.h")
    message(FATAL_ERROR "Generate and compile x650_standard before enabling the C++ replay")
  endif()
  # Reconfiguration may select a different SQP/RTI capsule with the same name.
  # Never combine its new identity header with a previously cached library.
  unset(X650_GENERATED_SOLVER CACHE)
  unset(X650_GENERATED_SOLVER)
  unset(X650_ACADOS CACHE)
  unset(X650_ACADOS)
  find_library(X650_GENERATED_SOLVER acados_ocp_solver_x650_standard PATHS "${X650_ACADOS_GENERATED_DIR}" NO_DEFAULT_PATH REQUIRED)
  find_library(X650_ACADOS acados PATHS "${ACADOS_SOURCE_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(x650_standard_acados_backend STATIC native/acados_backend.cpp)
  target_include_directories(x650_standard_acados_backend PRIVATE
    "${X650_ACADOS_GENERATED_DIR}" "${ACADOS_SOURCE_DIR}/include" "${ACADOS_SOURCE_DIR}/include/blasfeo/include" "${ACADOS_SOURCE_DIR}/include/hpipm/include"
    ../x650_wholebody_v1/benchmark/include)
  target_link_libraries(x650_standard_acados_backend PUBLIC x650_standard_case_builder "${X650_GENERATED_SOLVER}" "${X650_ACADOS}" PkgConfig::JSONCPP OpenSSL::Crypto)
  add_executable(x650_standard_solver_replay native/solver_replay.cpp)
  target_include_directories(x650_standard_solver_replay PRIVATE ../x650_wholebody_v1/benchmark/include "${X650_ACADOS_GENERATED_DIR}")
  target_link_libraries(x650_standard_solver_replay PRIVATE x650_standard_acados_backend PkgConfig::JSONCPP OpenSSL::Crypto)
  add_executable(x650_standard_native_replay native/acados_replay.cpp)
  target_compile_features(x650_standard_native_replay PRIVATE cxx_std_17)
  target_include_directories(x650_standard_native_replay PRIVATE
    "${X650_ACADOS_GENERATED_DIR}" "${ACADOS_SOURCE_DIR}/include" "${ACADOS_SOURCE_DIR}/include/blasfeo/include" "${ACADOS_SOURCE_DIR}/include/hpipm/include"
    ../x650_wholebody_v1/benchmark/include ../x650_wholebody_v1/native)
  target_link_libraries(x650_standard_native_replay PRIVATE "${X650_GENERATED_SOLVER}" "${X650_ACADOS}" PkgConfig::JSONCPP OpenSSL::Crypto)
endif()
