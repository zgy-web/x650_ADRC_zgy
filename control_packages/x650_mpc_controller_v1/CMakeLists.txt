cmake_minimum_required(VERSION 3.16)
project(x650_mpc_controller VERSION 1.0.0 LANGUAGES CXX)

set(X650_BACKEND_LIBRARY_DIR "" CACHE PATH "Existing ABI4 core / ABI2 optimizer directory")
set(X650_TS20_LIBRARY "" CACHE FILEPATH "Existing libts20_prediction_v3.so")
set(X650_NATIVE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../x650_wholebody_v1/native" CACHE PATH "Matching ABI4 headers")
option(X650_MPC_BUILD_REPLAY "Build JSON historical replay example (JsonCpp/OpenSSL)" ON)
option(X650_MPC_BUILD_COMMON_GATE "Build new common admission adapter (requires additive event-validator core symbol)" OFF)
option(X650_MPC_BUILD_HARDWARE_REPLAY "Offline PX4 message/CDR to original MPC to setpoint replay (no I/O)" OFF)
set(X650_MPC_COMMON_GATE_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/common_gate_recorded_v1.json" CACHE FILEPATH "SHA-bound two-case real historical common-gate fixture; empty explicitly disables recorded test")
set(X650_MPC_COMMON_GATE_FIXTURE_SHA256 "566404fbf4f974d85a9b947a4df7ba655d9c63c1930a498d665484c38edf70fc" CACHE STRING "Expected fixture SHA256")
foreach(file libx650_mpc_core.so libx650_optimization.so)
  if(NOT EXISTS "${X650_BACKEND_LIBRARY_DIR}/${file}")
    message(FATAL_ERROR "Provide existing verified libraries using X650_BACKEND_LIBRARY_DIR: ${file}")
  endif()
endforeach()
if(NOT EXISTS "${X650_TS20_LIBRARY}")
  message(FATAL_ERROR "Provide existing predictor using X650_TS20_LIBRARY")
endif()
add_library(x650_existing_core SHARED IMPORTED)
set_target_properties(x650_existing_core PROPERTIES IMPORTED_LOCATION "${X650_BACKEND_LIBRARY_DIR}/libx650_mpc_core.so")
add_library(x650_existing_optimization SHARED IMPORTED)
set_target_properties(x650_existing_optimization PROPERTIES IMPORTED_LOCATION "${X650_BACKEND_LIBRARY_DIR}/libx650_optimization.so")
add_library(x650_existing_predictor SHARED IMPORTED)
set_target_properties(x650_existing_predictor PROPERTIES IMPORTED_LOCATION "${X650_TS20_LIBRARY}")
add_library(x650_mpc_controller STATIC src/controller.cpp)
add_library(x650_mpc::controller ALIAS x650_mpc_controller)
target_include_directories(x650_mpc_controller PUBLIC include "${X650_NATIVE_INCLUDE_DIR}")
target_compile_features(x650_mpc_controller PUBLIC cxx_std_17)
target_compile_options(x650_mpc_controller PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(x650_mpc_controller PUBLIC x650_existing_core x650_existing_optimization)
if(X650_MPC_BUILD_COMMON_GATE)
  add_library(x650_mpc_common_gate STATIC src/common_gate.cpp)
  add_library(x650_mpc::common_gate ALIAS x650_mpc_common_gate)
  target_compile_features(x650_mpc_common_gate PUBLIC cxx_std_17)
  target_compile_options(x650_mpc_common_gate PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(x650_mpc_common_gate PUBLIC x650_mpc_controller)
  add_executable(x650_mpc_common_gate_one_step examples/common_gate_one_step.cpp)
  target_include_directories(x650_mpc_common_gate_one_step PRIVATE examples)
  target_link_libraries(x650_mpc_common_gate_one_step PRIVATE x650_mpc_common_gate x650_existing_predictor)
endif()
add_executable(x650_mpc_one_step examples/one_step.cpp)
target_link_libraries(x650_mpc_one_step PRIVATE x650_mpc_controller x650_existing_predictor)
target_include_directories(x650_mpc_one_step PRIVATE examples)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_x650_mpc_controller tests/test_controller.cpp)
  target_include_directories(test_x650_mpc_controller PRIVATE examples)
  target_link_libraries(test_x650_mpc_controller PRIVATE x650_mpc_controller x650_existing_predictor)
  foreach(test prepare_discard equivalence stale_tokens bad_history native_error wrong_publication reset)
    add_test(NAME mpc_interface_${test} COMMAND test_x650_mpc_controller ${test})
  endforeach()
  if(X650_MPC_BUILD_COMMON_GATE)
    find_package(Threads REQUIRED)
    add_executable(test_x650_mpc_common_gate tests/test_common_gate.cpp)
    target_include_directories(test_x650_mpc_common_gate PRIVATE examples)
    target_link_libraries(test_x650_mpc_common_gate PRIVATE x650_mpc_common_gate x650_existing_predictor Threads::Threads)
    foreach(test deadline_precheck safe_transaction wrong_publication native_rejection pending_reset deadline_after_validation allocation_cleanup)
      add_test(NAME mpc_common_gate_${test} COMMAND test_x650_mpc_common_gate ${test})
    endforeach()
    if(X650_MPC_COMMON_GATE_FIXTURE)
      if(NOT EXISTS "${X650_MPC_COMMON_GATE_FIXTURE}" OR NOT X650_MPC_COMMON_GATE_FIXTURE_SHA256 MATCHES "^[a-f0-9]+$")
        message(FATAL_ERROR "Provide existing common-gate fixture and expected lowercase SHA256")
      endif()
      find_package(PkgConfig REQUIRED)
      pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
      find_package(OpenSSL REQUIRED)
      add_executable(test_x650_mpc_common_gate_recorded tests/test_common_gate_recorded.cpp)
      target_include_directories(test_x650_mpc_common_gate_recorded PRIVATE ../x650_wholebody_v1/benchmark/include)
      target_link_libraries(test_x650_mpc_common_gate_recorded PRIVATE x650_mpc_common_gate x650_existing_predictor PkgConfig::JSONCPP OpenSSL::Crypto)
      add_test(NAME mpc_common_gate_historical_stopping COMMAND test_x650_mpc_common_gate_recorded "${X650_MPC_COMMON_GATE_FIXTURE}" "${X650_MPC_COMMON_GATE_FIXTURE_SHA256}")
    endif()
  endif()
endif()
if(X650_MPC_BUILD_REPLAY)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
  find_package(OpenSSL REQUIRED)
  add_executable(x650_mpc_replay examples/replay.cpp)
  target_include_directories(x650_mpc_replay PRIVATE ../x650_wholebody_v1/benchmark/include)
  target_link_libraries(x650_mpc_replay PRIVATE x650_mpc_controller x650_existing_predictor PkgConfig::JSONCPP OpenSSL::Crypto ${CMAKE_DL_LIBS})
endif()

if(X650_MPC_BUILD_HARDWARE_REPLAY)
  file(SHA256 "${X650_NATIVE_INCLUDE_DIR}/mpc_core.hpp" X650_REPLAY_HEADER_SHA256)
  if(NOT X650_REPLAY_HEADER_SHA256 STREQUAL "3ce12c395eb047167b5eda8fefb502fe56622e610cdfd3e82a4302f4a53c0093")
    message(FATAL_ERROR "Hardware replay is bound to the verified 2026-10-27 ABI4 binaries. Set X650_NATIVE_INCLUDE_DIR to compat/abi4_20261027 and verify the library hashes; ABI number alone is insufficient.")
  endif()
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(px4_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
  find_package(OpenSSL REQUIRED)
  add_executable(x650_hardware_replay examples/hardware_replay.cpp)
  target_include_directories(x650_hardware_replay PRIVATE
    ../x650_wholebody_v1/benchmark/include ../../ros2_ws/src/flyarm_hardware_state/include)
  target_link_libraries(x650_hardware_replay x650_mpc_controller x650_existing_predictor PkgConfig::JSONCPP OpenSSL::Crypto ${CMAKE_DL_LIBS})
  ament_target_dependencies(x650_hardware_replay rclcpp px4_msgs sensor_msgs)
  if(BUILD_TESTING)
    add_executable(test_x650_px4_adapter tests/test_px4_adapter.cpp)
    target_include_directories(test_x650_px4_adapter PRIVATE ../../ros2_ws/src/flyarm_hardware_state/include)
    target_link_libraries(test_x650_px4_adapter x650_mpc_controller)
    ament_target_dependencies(test_x650_px4_adapter rclcpp px4_msgs sensor_msgs)
    add_test(NAME mpc_px4_adapter_contract COMMAND test_x650_px4_adapter)
  endif()
endif()
