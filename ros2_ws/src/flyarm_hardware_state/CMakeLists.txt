cmake_minimum_required(VERSION 3.16)
project(flyarm_hardware_state)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
add_executable(px4_state_probe src/px4_state_probe.cpp)
target_compile_features(px4_state_probe PRIVATE cxx_std_17)
target_compile_options(px4_state_probe PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(px4_state_probe PRIVATE include)
ament_target_dependencies(px4_state_probe rclcpp px4_msgs)
install(TARGETS px4_state_probe DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(state_math_test test/state_math_test.cpp)
  target_compile_features(state_math_test PRIVATE cxx_std_17)
  target_include_directories(state_math_test PRIVATE include)
  add_test(NAME hardware_state_frames_and_time COMMAND state_math_test)
endif()
ament_package()
