cmake_minimum_required(VERSION 3.16)
project(flyarm_nmpc_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flyarm_control_interfaces REQUIRED)
find_package(flyarm_control_gateway REQUIRED)
find_package(flyarm_state_cpp REQUIRED)
find_package(flyarm_mpc_cpp REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(Threads REQUIRED)
get_filename_component(DEFAULT_CONTROL "${CMAKE_CURRENT_SOURCE_DIR}/../../../control_packages" ABSOLUTE)
set(X650_NATIVE_SOURCE_DIR "${DEFAULT_CONTROL}/x650_wholebody_v1/native" CACHE PATH "Matching ABI4 headers")
set(X650_NMPC_INCLUDE_DIR "${DEFAULT_CONTROL}/x650_nmpc_controller_v1/include" CACHE PATH "New online NMPC headers")
set(X650_MPC_CALLABLE_SOURCE_DIR "${DEFAULT_CONTROL}/x650_mpc_controller_v1" CACHE PATH "Retained callable MPC facade")
set(X650_MPC_CORE_LIBRARY "" CACHE FILEPATH "Existing ABI4 complete MPC core")
set(X650_NMPC_LIBRARY "" CACHE FILEPATH "New NMPC static library with online_solver")
if(NOT EXISTS "${X650_MPC_CORE_LIBRARY}" OR NOT EXISTS "${X650_NMPC_LIBRARY}" OR NOT EXISTS "${X650_NMPC_INCLUDE_DIR}/x650_nmpc/online_solver.hpp")
  message(FATAL_ERROR "Build new NMPC API first; provide X650_MPC_CORE_LIBRARY, X650_NMPC_LIBRARY, matching headers")
endif()
get_filename_component(CORE_LIBRARY_DIR "${X650_MPC_CORE_LIBRARY}" DIRECTORY)
find_path(MPC_HEADERS flyarm_mpc_cpp/config_io.hpp HINTS "${flyarm_mpc_cpp_DIR}/../../../include" REQUIRED)
find_path(GATEWAY_HEADERS flyarm_control_gateway/runtime_wire.hpp HINTS "${flyarm_control_gateway_DIR}/../../../include" REQUIRED)
find_path(STATE_HEADERS flyarm_state_cpp/source_core.hpp HINTS "${flyarm_state_cpp_DIR}/../../../include" REQUIRED)
add_executable(isolated_worker src/isolated_worker.cpp "${X650_MPC_CALLABLE_SOURCE_DIR}/src/controller.cpp")
target_compile_features(isolated_worker PRIVATE cxx_std_17)
# The exact committed original worker contract shadows unrelated source-tree
# PIL experiments. Other shared utilities come from the installed MPC package.
target_include_directories(isolated_worker PRIVATE frozen_include include "${X650_NATIVE_SOURCE_DIR}"
  "${X650_NMPC_INCLUDE_DIR}" "${X650_MPC_CALLABLE_SOURCE_DIR}/include" "${MPC_HEADERS}" "${GATEWAY_HEADERS}" "${STATE_HEADERS}")
target_compile_options(isolated_worker PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(isolated_worker rclcpp flyarm_control_interfaces)
target_link_libraries(isolated_worker "${X650_NMPC_LIBRARY}" "${X650_MPC_CORE_LIBRARY}"
  "${CORE_LIBRARY_DIR}/libx650_optimization.so" PkgConfig::JSONCPP OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(isolated_worker PROPERTIES INSTALL_RPATH "${CORE_LIBRARY_DIR}")
install(TARGETS isolated_worker DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY frozen_include/ DESTINATION share/${PROJECT_NAME}/frozen_include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_nmpc_diagnostics test/test_diagnostics.cpp)
  target_compile_features(test_nmpc_diagnostics PRIVATE cxx_std_17)
  target_include_directories(test_nmpc_diagnostics PRIVATE include "${X650_NATIVE_SOURCE_DIR}" "${X650_NMPC_INCLUDE_DIR}" "${MPC_HEADERS}" "${STATE_HEADERS}" "${GATEWAY_HEADERS}")
  target_link_libraries(test_nmpc_diagnostics PkgConfig::JSONCPP OpenSSL::Crypto)
  add_test(NAME nmpc_worker_distinct_objective_and_rejection_evidence COMMAND test_nmpc_diagnostics)
  add_executable(test_send_commit test/test_send_commit.cpp)
  target_compile_features(test_send_commit PRIVATE cxx_std_17)
  target_include_directories(test_send_commit PRIVATE include)
  add_test(NAME upper_worker_actual_send_commit_and_callback_order COMMAND test_send_commit)
  add_executable(test_fast_json test/test_fast_json.cpp)
  target_compile_features(test_fast_json PRIVATE cxx_std_17)
  target_include_directories(test_fast_json PRIVATE include)
  target_link_libraries(test_fast_json PkgConfig::JSONCPP)
  add_test(NAME upper_worker_lossless_json_numeric_and_string_roundtrip COMMAND test_fast_json)
endif()
ament_package()
