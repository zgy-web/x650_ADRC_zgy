cmake_minimum_required(VERSION 3.16)
project(flyarm_mpc_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flyarm_control_interfaces REQUIRED)
find_package(flyarm_control_gateway REQUIRED)
find_package(flyarm_state_cpp REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(Threads REQUIRED)
get_filename_component(DEFAULT_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../../control_packages/x650_wholebody_v1/native" ABSOLUTE)
set(X650_NATIVE_SOURCE_DIR "${DEFAULT_NATIVE}" CACHE PATH "Explicit complete C++ MPC source headers")
set(X650_MPC_CORE_LIBRARY "" CACHE FILEPATH "Explicit already built complete C++ MPC library")
if(NOT EXISTS "${X650_MPC_CORE_LIBRARY}" OR NOT EXISTS "${X650_NATIVE_SOURCE_DIR}/mpc_core.hpp")
  message(FATAL_ERROR "Build/qualify complete core first and provide X650_MPC_CORE_LIBRARY plus matching headers")
endif()
find_path(GATEWAY_HEADERS flyarm_control_gateway/runtime_wire.hpp HINTS "${flyarm_control_gateway_DIR}/../../../include" REQUIRED)
find_path(STATE_HEADERS flyarm_state_cpp/source_core.hpp HINTS "${flyarm_state_cpp_DIR}/../../../include" REQUIRED)
add_executable(isolated_worker src/isolated_worker.cpp)
target_compile_features(isolated_worker PRIVATE cxx_std_17)
target_include_directories(isolated_worker PRIVATE include "${X650_NATIVE_SOURCE_DIR}" "${GATEWAY_HEADERS}" "${STATE_HEADERS}")
target_compile_options(isolated_worker PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(isolated_worker rclcpp flyarm_control_interfaces)
target_link_libraries(isolated_worker "${X650_MPC_CORE_LIBRARY}" PkgConfig::JSONCPP OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
get_filename_component(CORE_LIBRARY_DIR "${X650_MPC_CORE_LIBRARY}" DIRECTORY)
set_target_properties(isolated_worker PROPERTIES INSTALL_RPATH "${CORE_LIBRARY_DIR}")
install(TARGETS isolated_worker DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_worker_contract test/test_worker_contract.cpp)
  target_compile_features(test_worker_contract PRIVATE cxx_std_17)
  target_include_directories(test_worker_contract PRIVATE include "${X650_NATIVE_SOURCE_DIR}" "${GATEWAY_HEADERS}" "${STATE_HEADERS}")
  add_test(NAME cpp_worker_raw_contract COMMAND test_worker_contract)
  add_executable(test_config_diagnostics test/test_config_diagnostics.cpp)
  target_compile_features(test_config_diagnostics PRIVATE cxx_std_17)
  target_include_directories(test_config_diagnostics PRIVATE include "${X650_NATIVE_SOURCE_DIR}" "${STATE_HEADERS}")
  target_link_libraries(test_config_diagnostics PkgConfig::JSONCPP OpenSSL::Crypto Threads::Threads)
  add_test(NAME cpp_worker_startup_and_diagnostics COMMAND test_config_diagnostics)
  foreach(name test_normal_contract test_normal_config trajectory_probe)
    add_executable(${name} test/${name}.cpp)
    target_compile_features(${name} PRIVATE cxx_std_17)
    target_include_directories(${name} PRIVATE include "${X650_NATIVE_SOURCE_DIR}" "${GATEWAY_HEADERS}" "${STATE_HEADERS}")
  endforeach()
  target_link_libraries(test_normal_config PkgConfig::JSONCPP OpenSSL::Crypto Threads::Threads)
  add_test(NAME cpp_worker_normal_contract COMMAND test_normal_contract)
  add_test(NAME cpp_worker_normal_configuration COMMAND test_normal_config)
endif()
ament_package()
