cmake_minimum_required(VERSION 3.8)
project(flyarm_control_gateway)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flyarm_control_interfaces REQUIRED)
add_executable(shadow_gateway src/shadow_gateway.cpp)
target_compile_features(shadow_gateway PUBLIC cxx_std_17)
target_include_directories(shadow_gateway PRIVATE include)
ament_target_dependencies(shadow_gateway rclcpp flyarm_control_interfaces)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(shadow_gateway PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS shadow_gateway DESTINATION lib/${PROJECT_NAME})
foreach(runtime_node state_ingress periodic_dispatcher consumption_source)
  add_executable(${runtime_node} src/${runtime_node}.cpp)
  target_compile_features(${runtime_node} PUBLIC cxx_std_17)
  target_include_directories(${runtime_node} PRIVATE include)
  ament_target_dependencies(${runtime_node} rclcpp flyarm_control_interfaces)
  target_compile_options(${runtime_node} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${runtime_node} DESTINATION lib/${PROJECT_NAME})
endforeach()
set(MAVLINK_INCLUDE_DIR "" CACHE PATH "Existing PX4 generated MAVLink include directory")
if(MAVLINK_INCLUDE_DIR)
  find_package(std_msgs REQUIRED)
  add_executable(live_dispatcher src/periodic_dispatcher.cpp)
  target_compile_features(live_dispatcher PUBLIC cxx_std_17)
  target_compile_definitions(live_dispatcher PRIVATE FLYARM_LIVE_SIM=1)
  target_include_directories(live_dispatcher PRIVATE include ${MAVLINK_INCLUDE_DIR})
  ament_target_dependencies(live_dispatcher rclcpp flyarm_control_interfaces std_msgs)
  install(TARGETS live_dispatcher DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_shadow_gate test/test_shadow_gate.cpp)
  target_compile_features(test_shadow_gate PUBLIC cxx_std_17)
  target_include_directories(test_shadow_gate PRIVATE include)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_shadow_gate PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME shadow_gate_synthetic COMMAND test_shadow_gate)
  add_executable(test_periodic_gate test/test_periodic_gate.cpp)
  target_compile_features(test_periodic_gate PUBLIC cxx_std_17)
  target_include_directories(test_periodic_gate PRIVATE include)
  add_test(NAME periodic_gate_contract COMMAND test_periodic_gate)
  add_executable(test_consumption_csv test/test_consumption_csv.cpp)
  target_compile_features(test_consumption_csv PUBLIC cxx_std_17)
  target_include_directories(test_consumption_csv PRIVATE include)
  add_test(NAME consumption_csv_contract COMMAND test_consumption_csv)
  add_executable(test_consumption_lifecycle test/test_consumption_lifecycle.cpp)
  target_compile_features(test_consumption_lifecycle PUBLIC cxx_std_17)
  target_include_directories(test_consumption_lifecycle PRIVATE include)
  add_test(NAME consumption_startup_lifecycle COMMAND test_consumption_lifecycle)
  add_executable(test_admission_cadence test/test_admission_cadence.cpp)
  target_compile_features(test_admission_cadence PUBLIC cxx_std_17)
  target_include_directories(test_admission_cadence PRIVATE include)
  add_test(NAME actual_source_cadence COMMAND test_admission_cadence)
endif()
ament_package()
