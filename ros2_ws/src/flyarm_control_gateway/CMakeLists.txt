cmake_minimum_required(VERSION 3.8)
project(flyarm_control_gateway)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flyarm_control_interfaces REQUIRED)
add_executable(shadow_gateway src/shadow_gateway.cpp)
target_compile_features(shadow_gateway PUBLIC cxx_std_17)
target_include_directories(shadow_gateway PRIVATE include)
ament_target_dependencies(shadow_gateway rclcpp flyarm_control_interfaces)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(shadow_gateway PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS shadow_gateway DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_shadow_gate test/test_shadow_gate.cpp)
  target_compile_features(test_shadow_gate PUBLIC cxx_std_17)
  target_include_directories(test_shadow_gate PRIVATE include)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_shadow_gate PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME shadow_gate_synthetic COMMAND test_shadow_gate)
endif()
ament_package()
