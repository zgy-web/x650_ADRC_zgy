cmake_minimum_required(VERSION 3.8)
project(flyarm_state_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(flyarm_control_interfaces REQUIRED)
add_executable(state_source src/state_source.cpp)
target_compile_features(state_source PUBLIC cxx_std_17)
target_include_directories(state_source PRIVATE include)
target_compile_options(state_source PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(state_source rclcpp px4_msgs rosgraph_msgs flyarm_control_interfaces)
install(TARGETS state_source DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_source_core test/test_source_core.cpp)
  target_compile_features(test_source_core PUBLIC cxx_std_17)
  target_include_directories(test_source_core PRIVATE include)
  target_compile_options(test_source_core PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME raw_state_causal_source COMMAND test_source_core)
  add_executable(replay_source_core test/replay_source_core.cpp)
  target_compile_features(replay_source_core PUBLIC cxx_std_17)
  target_include_directories(replay_source_core PRIVATE include)
  target_compile_options(replay_source_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
ament_package()
